297 def check_for_updates(self):
298 """Check for client updates"""
299 def fetch():
300 try:
301 ctx = get_ssl_context()
302 req = urllib.request.Request(
303 CONFIG["version_url"],
304 headers={"User-Agent": f"AIKOLauncher/{CONFIG['version']}"}
305 )
306 with urllib.request.urlopen(req, timeout=10, context=ctx) as r:
307 data = json.loads(r.read().decode('utf-8'))
308 self.root.after(0, lambda: self.process_version_info(data))
309 except Exception as e:
310 print(f"Update check failed: {e}")
311 self.root.after(0, lambda: self.update_news_default())
312
313 threading.Thread(target=fetch, daemon=True).start()

421 def auto_apply_update(self):
422 """Automatically apply update without asking"""
423 if not self.update_available or not self.update_info or self.downloading:
424 return
425
426 # Check for patches first
427 if "patches" in self.update_info and self.update_info["patches"]:
428 local_build = 0
429 if self.local_version and "client" in self.local_version:
430 local_build = self.local_version["client"].get("build", 0)
431
432 applicable_patches = [
433 p for p in self.update_info["patches"]
434 if p.get("from_build", 0) == local_build
435 ]
436
437 if applicable_patches:
438 self.download_patch(applicable_patches[0])
439 return
440
441 # Full update - automatic
442 client_info = self.update_info["client"]
443 self.download_update(client_info)

445 def apply_update(self):
446 """Apply available update (manual trigger)"""
447 if not self.update_available or not self.update_info:
448 return
449
450 # Check for patches first
451 if "patches" in self.update_info and self.update_info["patches"]:
452 # Find applicable patches
453 local_build = 0
454 if self.local_version and "client" in self.local_version:
455 local_build = self.local_version["client"].get("build", 0)
456
457 applicable_patches = [
458 p for p in self.update_info["patches"]
459 if p.get("from_build", 0) == local_build
460 ]
461
462 if applicable_patches:
463 # Apply patch instead of full download
464 patch = applicable_patches[0]
465 result = messagebox.askyesno(
466 "Mise a jour",
467 f"Appliquer le patch?\n\n"
468 f"Version: {patch.get('version', '?')}\n"
469 f"Taille: ~{patch.get('size_mb', '?')} MB\n\n"
470 "Continuer?"
471 )
472 if result:
473 self.download_patch(patch)

475
476 # Full update
477 client_info = self.update_info["client"]
478 result = messagebox.askyesno(

484 )
485 if result:
486 self.download_update(client_info)

488 def download_patch(self, patch_info):
489 """Download and apply a patch"""
490 self.downloading = True
491 self.update_frame.pack_forget()
492 self.play_btn.config(text="...", state="disabled")
493 self.progress_frame.pack(fill="x", pady=(0, 10))
494 self.progress_text.config(text="Telechargement du patch...")
495 self.progress_bar["value"] = 0
496
497 def download():
498 try:
499 url = patch_info["url"]
500 patch_path = self.install_path / "patch.zip"
501
502 ctx = get_ssl_context()
503 req = urllib.request.Request(url, headers={
504 "User-Agent": f"AIKOLauncher/{CONFIG['version']}"
505 })
506
507 response = urllib.request.urlopen(req, timeout=60, context=ctx)
508 total = int(response.headers.get('content-length', 0))
509 if total == 0:
510 total = patch_info.get("size_mb", 10) * 1024 * 1024
511
512 downloaded = 0
513 block_size = 1024 * 256
514
515 with open(patch_path, 'wb') as f:
516 while True:
517 block = response.read(block_size)
518 if not block:
519 break
520 f.write(block)
521 downloaded += len(block)
522 pct = min(100, (downloaded / total) * 100)
523 mb = downloaded / (1024 * 1024)
524 total_mb = total / (1024 * 1024)
525 self.root.after(0, lambda p=pct, m=mb, t=total_mb: self.update_progress(p, m, t))
526
527 response.close()
528
529 # Verify hash if provided (auto-detect MD5 vs SHA256)
530 if "hash" in patch_info and patch_info["hash"]:
531 self.update_ui("Verification du patch...")
532 with open(patch_path, "rb") as f:
533 content = f.read()
534 h = patch_info["hash"]
535 if len(h) == 64: # SHA256
536 file_hash = hashlib.sha256(content).hexdigest()
537 else: # MD5
538 file_hash = hashlib.md5(content).hexdigest()
539 if file_hash != patch_info["hash"]:
540 self.root.after(0, lambda: self.download_error("Hash invalide - fichier corrompu"))
541 patch_path.unlink()
542 return
543
544 # Extract patch
545 self.update_ui("Application du patch...")
546 with zipfile.ZipFile(patch_path, 'r') as zf:
547 zf.extractall(self.install_path)
548
549 patch_path.unlink()
550
551 # Update local version
552 new_version = {
553 "client": {
554 "version": patch_info.get("version", "1.0.1"),
555 "build": patch_info.get("to_build", 1),
556 "date": patch_info.get("date", "")
557 }
558 }
559 self.save_local_version(new_version)
560 self.local_version = new_version
561
562 self.root.after(0, self.update_complete)
563
564 except Exception as e:
565 self.root.after(0, lambda: self.download_error(str(e)))
566
567 threading.Thread(target=download, daemon=True).start()

616 if len(h) == 64: # SHA256
617 file_hash = hashlib.sha256(content).hexdigest()
618 else: # MD5
619 file_hash = hashlib.md5(content).hexdigest()
620 if file_hash != client_info["hash"]:
621 self.root.after(0, lambda: self.download_error("Hash invalide - fichier corrompu"))
622 zip_path.unlink()
623 return
624
625 # Extract
626 self.update_ui("Installation de la mise a jour...")
627 with zipfile.ZipFile(zip_path, 'r') as zf:
628 names = zf.namelist()
629 if names and '/' in names[0]:
630 root_folder = names[0].split('/')[0]
631 else:
632 root_folder = None
633
634 zf.extractall(self.install_path)
635
636 if root_folder:
637 subfolder = self.install_path / root_folder
638 if subfolder.exists() and subfolder.is_dir():
639 for item in subfolder.iterdir():
640 dest = self.install_path / item.name
641 if dest.exists():
642 if dest.is_file():
643 dest.unlink()
644 elif dest.is_dir():
645 import shutil
646 shutil.rmtree(dest)
647 item.rename(dest)
648 try:
649 subfolder.rmdir()
650 except:
651 pass
652
653 zip_path.unlink()
654
655 # Update local version
656 new_version = {
657 "client": {
658 "version": client_info.get("version", "1.0.0"),
659 "build": client_info.get("build", 1),
660 "date": client_info.get("date", "")
661 }
662 }
663 self.save_local_version(new_version)
664 self.local_version = new_version
665
666 self.root.after(0, self.update_complete)
667
668 except Exception as e:
669 self.root.after(0, lambda: self.download_error(str(e)))
